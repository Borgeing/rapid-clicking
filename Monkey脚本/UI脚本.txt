    const DRAG_THRESHOLD = 3;

    /************* 内部变量 *************/
    let targetElement = null;
    let clickingTimer = null;
    let isPickingMode = false;
    let batchCount = 0;
    let overlay = null;
    let lastPickedSelector = '';
    let isPanelMinimized = false;

            document.getElementById('intervalInput').value = CLICK_INTERVAL_MS;
            document.getElementById('batchInput').value = MAX_CLICK_PER_BATCH;
            document.getElementById('restInput').value = REST_AFTER_BATCH_MS;
        });

            CLICK_INTERVAL_MS = parseInt(document.getElementById('intervalInput').value) || 300;
            MAX_CLICK_PER_BATCH = parseInt(document.getElementById('batchInput').value) || 50;
            REST_AFTER_BATCH_MS = parseInt(document.getElementById('restInput').value) || 2000;
            showMessage('参数已保存，立即生效');
            panel.style.display = 'none';
        });
    }

    /************* 流畅拖拽逻辑 *************/
    function smoothDragModel(dragElement) {
        let isDragging = false;
        let startX = 0;
        let startY = 0;
        let elementX = 0;
        let elementY = 0;
        let dragStartDistance = 0;
        let animationFrameId = null;

        dragElement.addEventListener('mousedown', (e) => {
            if (e.button !== 0 || e.ctrlKey || e.metaKey || e.shiftKey) return;

            e.preventDefault();
            e.stopPropagation();

            startX = e.clientX;
            startY = e.clientY;
            elementX = dragElement.offsetLeft;
            elementY = dragElement.offsetTop;

            document.body.classList.add('dragging');
            isDragging = false;

            document.addEventListener('mousemove', handleMouseMove);
            document.addEventListener('mouseup', handleMouseUp);
        });

        function handleMouseMove(e) {
            if (!document.body.classList.contains('dragging')) return;

            const deltaX = e.clientX - startX;
            const deltaY = e.clientY - startY;
            dragStartDistance = Math.sqrt(deltaX * deltaX + deltaY * deltaY);

            if (!isDragging) {
                if (dragStartDistance > DRAG_THRESHOLD) {
                    isDragging = true;
                } else {
                    return;
                }
            }

            if (animationFrameId) cancelAnimationFrame(animationFrameId);
            animationFrameId = requestAnimationFrame(() => {
                let newLeft = elementX + deltaX;
                let newTop = elementY + deltaY;

                newLeft = Math.max(0, Math.min(newLeft, window.innerWidth - dragElement.offsetWidth));
                newTop = Math.max(0, Math.min(newTop, window.innerHeight - dragElement.offsetHeight));

                dragElement.style.left = `${newLeft}px`;
                dragElement.style.top = `${newTop}px`;
            });
        }

        function handleMouseUp() {
            if (animationFrameId) cancelAnimationFrame(animationFrameId);
            document.removeEventListener('mousemove', handleMouseMove);
            document.removeEventListener('mouseup', handleMouseUp);

            document.body.classList.remove('dragging');
            isDragging = false;
            dragStartDistance = 0;
        }
    }

    /************* 核心功能 *************/
    function showMessage(msg, timeout = 2000) {
        console.log('[连点器]', msg);
        let div = document.createElement('div');
        div.textContent = msg;
        Object.assign(div.style, {
            position: 'fixed',
            zIndex: 999999,
            left: '50%',
            top: '20px',
            transform: 'translateX(-50%)',
            background: 'rgba(0,0,0,0.75)',
            color: '#fff',
            padding: '6px 12px',
            borderRadius: '4px',
            fontSize: '12px',
            fontFamily: 'sans-serif',
            pointerEvents: 'none'
        });
        document.body.appendChild(div);
        setTimeout(() => div.remove(), timeout);
    }

    function isHotkey(e, conf) {
        return !!conf && e.key.toLowerCase() === conf.key.toLowerCase() &&
            !!e.ctrlKey === !!conf.ctrlKey && !!e.altKey === !!conf.altKey &&
            !!e.shiftKey === !!conf.shiftKey;
    }

    function startClicking() {
        if (!targetElement || !document.body.contains(targetElement)) {
            showMessage('未找到目标元素，请先按 Ctrl+Alt+C 选择元素');
            stopClicking();
            return;
        }
        if (clickingTimer) return;
        showMessage('连点开始');
        batchCount = 0;
        clickingTimer = setInterval(() => {
            if (!targetElement || !document.body.contains(targetElement)) {
                showMessage('目标元素已不存在，停止连点');
                stopClicking();
                return;
            }
            try {
                targetElement.click();
                batchCount++;
                if (batchCount >= MAX_CLICK_PER_BATCH) {
                    clearInterval(clickingTimer);
                    clickingTimer = null;
                    showMessage(`本轮已点击 ${MAX_CLICK_PER_BATCH} 次，休息中...`);
                    setTimeout(() => { batchCount = 0; startClicking(); }, REST_AFTER_BATCH_MS);
                }
            } catch (err) {
                console.error('[连点器] 点击出错：', err);
            }
        }, CLICK_INTERVAL_MS);
    }

    function stopClicking() {
        if (clickingTimer) {
            clearInterval(clickingTimer);
            clickingTimer = null;
            showMessage('连点已停止');
        }
    }

    // 选择元素模式
    function createOverlay() {
        if (overlay) return overlay;
        overlay = document.createElement('div');
        Object.assign(overlay.style, {
            position: 'absolute',
            border: '2px solid #ff9800',
            background: 'rgba(255,152,0,0.15)',
            pointerEvents: 'none',
            zIndex: 999998
        });
        document.body.appendChild(overlay);
        return overlay;
    }

    function highlightElement(el) {
        if (!el) return;
        const rect = el.getBoundingClientRect();
        const ov = createOverlay();
        Object.assign(ov.style, {
            left: `${window.scrollX + rect.left}px`,
            top: `${window.scrollY + rect.top}px`,
            width: `${rect.width}px`,
            height: `${rect.height}px`,
            display: 'block'
        });
    }

    function hideOverlay() {
        if (overlay) overlay.style.display = 'none';
    }

    function buildSimpleSelector(el) {
        if (!el || !el.tagName) return '';
        let selector = el.tagName.toLowerCase();
        if (el.id) selector += `#${el.id}`;
        if (el.classList && el.classList.length) {
            selector += '.' + Array.from(el.classList).slice(0, 3).join('.');
        }
        return selector;
    }

    function enablePickingMode() {
        if (isPickingMode) return;
        isPickingMode = true;
        showMessage('选择元素模式：移动鼠标到目标元素，点击一次即可选中（ESC 取消）', 4000);
        const moveHandler = (e) => highlightElement(e.target);
        const clickHandler = (e) => {
            e.preventDefault();
            e.stopPropagation();
            targetElement = e.target;
            lastPickedSelector = buildSimpleSelector(e.target);
            hideOverlay();
            disablePickingMode();
            showMessage(`已选中元素：${lastPickedSelector || e.target.tagName}, 自动开始连点`);
            startClicking();
        };
        const keyHandler = (e) => {
            if (e.key === 'Escape') {
                e.preventDefault();
                hideOverlay();
                disablePickingMode();
                showMessage('已取消选择元素模式');
            }
        };
        window.__autoClicker_pickHandlers = { moveHandler, clickHandler, keyHandler };
        document.addEventListener('mousemove', moveHandler, true);
        document.addEventListener('click', clickHandler, true);
        document.addEventListener('keydown', keyHandler, true);
    }

    function disablePickingMode() {
        if (!isPickingMode) return;
        isPickingMode = false;
        const handlers = window.__autoClicker_pickHandlers;
        if (handlers) {
            document.removeEventListener('mousemove', handlers.moveHandler, true);
            document.removeEventListener('click', handlers.clickHandler, true);
            document.removeEventListener('keydown', handlers.keyHandler, true);
        }
        window.__autoClicker_pickHandlers = null;
    }

    // 快捷键监听
    document.addEventListener('keydown', (e) => {
        const tag = (e.target.tagName || '').toLowerCase();
        if (tag === 'input' || tag === 'textarea' || e.target.isContentEditable) return;

        if (isHotkey(e, HOTKEY_TOGGLE)) {
            e.preventDefault();
            clickingTimer ? stopClicking() : startClicking();
            return;
        }

        if (isHotkey(e, HOTKEY_PICK)) {
            e.preventDefault();
            if (isPickingMode) {
                disablePickingMode();
                hideOverlay();
                showMessage('已退出选择元素模式');
            } else {
                stopClicking();
                enablePickingMode();
            }
        }
    });

    /************* 初始化 *************/
    createCircleAndPanel();
    showMessage('连点器已加载：点击圆环配置参数 | Ctrl+Alt+C 选元素 | Ctrl+Alt+S 启停', 5000);
})();
